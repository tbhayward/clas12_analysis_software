\subsection{Binning}

In order to study both the $-t'$ and $x_B$ dependence we bin the data in four separate bins in $x_B$: [0.1,0.25], [0.25,0.35], [0.35,0.45] and [0.45,0.60]  of which each is subdivided into $-t'$ bins: [0.05,0.25], [0.25,0.45], [0.45,0.65], [0.65,0.85], [0.85,1.05], [1.05,1.25]. For organizational purposes in this analysis note these bins may later be labeled as bins 1-24 and defined such that:

\begin{itemize}
  \item $0.10 < x_B < 0.25$ (bins 1--6):
    \begin{itemize}
      \item Bin 1: $0.05 < -t' < 0.25$ (GeV$^2$)
      \item Bin 2: $0.25 < -t' < 0.45$ (GeV$^2$)
      \item Bin 3: $0.45 < -t' < 0.65$ (GeV$^2$)
      \item Bin 4: $0.65 < -t' < 0.85$ (GeV$^2$)
      \item Bin 5: $0.85 < -t' < 1.05$ (GeV$^2$)
      \item Bin 6: $1.05 < -t' < 1.25$ (GeV$^2$)
    \end{itemize}
  \item $0.25 < x_B < 0.35$ (bins 7--12):
    \begin{itemize}
      \item Bin 7: $0.05 < -t' < 0.25$ (GeV$^2$)
      \item Bin 8: $0.25 < -t' < 0.45$ (GeV$^2$)
      \item Bin 9: $0.45 < -t' < 0.65$ (GeV$^2$)
      \item Bin 10: $0.65 < -t' < 0.85$ (GeV$^2$)
      \item Bin 11: $0.85 < -t' < 1.05$ (GeV$^2$)
      \item Bin 12: $1.05 < -t' < 1.25$ (GeV$^2$)
    \end{itemize}
  \item $0.35 < x_B < 0.45$ (bins 13--18):
    \begin{itemize}
      \item Bin 13: $0.05 < -t' < 0.25$ (GeV$^2$)
      \item Bin 14: $0.25 < -t' < 0.45$ (GeV$^2$)
      \item Bin 15: $0.45 < -t' < 0.65$ (GeV$^2$)
      \item Bin 16: $0.65 < -t' < 0.85$ (GeV$^2$)
      \item Bin 17: $0.85 < -t' < 1.05$ (GeV$^2$)
      \item Bin 18: $1.05 < -t' < 1.25$ (GeV$^2$)
    \end{itemize}
  \item $0.45 < x_B < 0.60$ (bins 19--24):
    \begin{itemize}
      \item Bin 19: $0.05 < -t' < 0.25$ (GeV$^2$)
      \item Bin 20: $0.25 < -t' < 0.45$ (GeV$^2$)
      \item Bin 21: $0.45 < -t' < 0.65$ (GeV$^2$)
      \item Bin 22: $0.65 < -t' < 0.85$ (GeV$^2$)
      \item Bin 23: $0.85 < -t' < 1.05$ (GeV$^2$)
      \item Bin 24: $1.05 < -t' < 1.25$ (GeV$^2$)
    \end{itemize}
\end{itemize}
